By default, ArchonWiki saves locally to IndexedDB and exports standalone .html snapshots. To let everyone visiting your GitHub Pages site share accounts, posts, follows, and blocks in real time, connect a free Supabase project:

Supabase SQL Setup (Run once in Supabase SQL Editor) 
create table if not exists archon_users (
  username text primary key,
  password_hash text not null,
  salt text not null,
  role text default 'Arch / Hyprland',
  bio text default '',
  following jsonb default '[]'::jsonb,
  blocked jsonb default '[]'::jsonb,
  bookmarks jsonb default '[]'::jsonb,
  created_at text
);

create table if not exists archon_articles (
  id text primary key,
  title text not null,
  category text not null,
  tags jsonb default '[]'::jsonb,
  author text not null,
  updated_at text,
  content text not null
);

alter table archon_users enable row level security;
alter table archon_articles enable row level security;
create policy "Public read/write users" on archon_users for all using (true) with check (true);
create policy "Public read/write articles" on archon_articles for all using (true) with check (true);
